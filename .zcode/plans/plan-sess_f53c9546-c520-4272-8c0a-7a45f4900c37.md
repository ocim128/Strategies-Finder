  ```ts
  type ReattachPollOptions<TStatus> = {
    isStillOwned: () => boolean;              // reattachPollingStopped OR activeXxxRunId === runId
    cancellableDelay: (ms: number) => Promise<void>;  // wraps the timer + resolve pair
    cancelDelay: () => void;                  // nulls timer/resolve
    poll: () => Promise<TStatus>;             // fetch + json + non-2xx→throw
    onHealthy: (status: TStatus) => "continue" | "done";
    onGiveUp: () => void;                     // DOM restore + persisted marker clear
    onTransient: (consecutive: number, max: number) => void;  // status text update
    failureBackoffMs?: readonly number[];     // default [2s,5s,10s,15s]
    maxConsecutiveFailures?: number;          // default 20
    healthyDelayMs?: number | ((pollIndex: number) => number); // server-run needs the 2s→5s step-down; TOP_MEAN is fixed 2s
  };
  ```
